41 void draw_network(const std::vector<models::layer> &layers) override {
42 if (layers.size() < MIN_LAYERS) {
43 return;
44 }
45
46 for (std::size_t i{0}; i < layers.size(); i++) {
47 if (i > 0) {
48 draw_connections(layers.at(i - 1), layers.at(i), i, layers.size());

50
51 if (i == 0) {

92 void draw_neurons(const models::layer &layer, std::size_t column,

119 void draw_connections(const models::layer &source, const models::layer &dest,
120 std::size_t column, std::size_t columns) {
121 const auto source_n{static_cast<std::size_t>(source.activations.size())};
122 const auto dest_n{static_cast<std::size_t>(dest.activations.size())};
123
124 if (source_n == 0 || dest_n == 0) {
125 return;
126 }
127
128 for (std::size_t j{0}; j < dest_n; j++) {
129 rank_incoming(source, dest, j);
130 if (candidates_.empty()) {
131 continue;
132 }
133
134 const auto strongest{candidates_.front().strength};
135 if (strongest <= 0.0f) {
136 continue;
137 }
138
139 const auto column_index{static_cast<Eigen::Index>(j)};
140 const auto to{neuron_position(column, j, dest_n, columns)};
141
142 for (const auto &candidate : candidates_) {
143 // A layer's weights are indexed [destination][source], since
144 // z = weights * previous_activations.
145 const auto weight{dest.weights(
146 column_index, static_cast<Eigen::Index>(candidate.source))};
147 const auto share{candidate.strength / strongest};
148 const auto hue{weight >= 0.0f ? POSITIVE_COLOR : NEGATIVE_COLOR};
149
150 renderer_.get().set_color(hue * (CONNECTION_GAIN * share));
151 renderer_.get().draw_line(
152 neuron_position(column - 1, candidate.source, source_n, columns),
153 to, CONNECTION_THICKNESS);
154 }
155 }
156 }

164 void rank_incoming(const models::layer &source, const models::layer &dest,
165 std::size_t destination) {
166 const auto column{static_cast<Eigen::Index>(destination)};
167 candidates_.clear();
168
169 for (Eigen::Index i{0}; i < source.activations.size(); i++) {
170 const auto strength{std::fabs(dest.weights(column, i)) *
171 std::fabs(source.activations(i))};
172
173 if (strength > 0.0f) {
174 candidates_.push_back(
175 {.strength = strength, .source = static_cast<std::size_t>(i)});
176 }
177 }
178
179 const auto keep{std::min(candidates_.size(), KEEP_PER_NEURON)};
180 std::partial_sort(candidates_.begin(),
181 candidates_.begin() + static_cast<std::ptrdiff_t>(keep),
182 candidates_.end(),
183 [](const candidate &l, const candidate &r) {
184 return l.strength > r.strength;
185 });
186 candidates_.resize(keep);
187 }

195 void draw_winner(const std::vector<models::layer> &layers) {
196 const auto &output{layers.back().activations};
197 const auto count{static_cast<std::size_t>(output.size())};
198 if (count == 0) {
199 return;
200 }
201
202 const auto last{layers.size() - 1};
203 const auto radius{std::fmin(spacing(count) * NEURON_FILL, MAX_RADIUS) *
204 WINNER_SCALE};
205
206 renderer_.get().set_color(WINNER_COLOR);
207 draw_ring(neuron_position(last, index_of_max(output), count, layers.size()),
208 radius);
209 }

221 std::size_t columns) {

244 [[nodiscard]] static float column_center(std::size_t column,
245 std::size_t columns) {
246 const auto others{columns > 1 ? columns - 1 : 1};
247 const auto input_width{diagram_span() * shared::INPUT_COLUMN_SHARE};
248 const auto rest_width{diagram_span() - input_width};
249 const auto hidden_width{rest_width / static_cast<float>(others)};
250
251 if (column == 0) {
252 return shared::NETWORK_LEFT + input_width * HALF;
253 }
254
255 return shared::NETWORK_LEFT + input_width +
256 hidden_width * (static_cast<float>(column - 1) + HALF);
257 }

263 [[nodiscard]] static Eigen::Vector2f input_pixel_center(std::size_t index) {
264 const auto side{input_side()};
265 const auto across{pixels_across()};
266 const auto cell{pixel_offset(index, across)};

268 const auto left{shared::NETWORK_LEFT};

272 bottom + (cell.y() + HALF) * side * shared::WINDOW_ASPECT / across};
